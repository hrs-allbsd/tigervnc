find_package(PkgConfig REQUIRED)
pkg_check_modules(SIXEL REQUIRED IMPORTED_TARGET libsixel)
add_executable(vncsixel main.cxx Terminal.cxx Connection.cxx Input.cxx Performance.cxx)
target_link_libraries(vncsixel PRIVATE rfbclient rfb network rdr core PkgConfig::SIXEL)
install(TARGETS vncsixel DESTINATION ${CMAKE_INSTALL_FULL_BINDIR})

if(GTest_FOUND)
  include(GoogleTest)
  enable_testing()
  add_executable(vncsixel-input-tests tests/InputTest.cxx tests/FramebufferTest.cxx Input.cxx)
  target_include_directories(vncsixel-input-tests PRIVATE ${CMAKE_SOURCE_DIR}/common ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(vncsixel-input-tests PRIVATE GTest::gtest_main)
  target_compile_options(vncsixel-input-tests PRIVATE -std=gnu++17)
  gtest_discover_tests(vncsixel-input-tests)
endif()
