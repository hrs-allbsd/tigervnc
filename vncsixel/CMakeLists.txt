find_package(PkgConfig REQUIRED)
pkg_check_modules(SIXEL REQUIRED IMPORTED_TARGET libsixel)
add_executable(vncsixel main.cxx Terminal.cxx Connection.cxx Input.cxx Performance.cxx HostCursor.cxx)
target_link_libraries(vncsixel PRIVATE rfbclient rfb network rdr core PkgConfig::SIXEL)
option(ENABLE_HOST_CURSOR "Enable optional local XFixes pointer control" ON)
if(ENABLE_HOST_CURSOR)
  pkg_check_modules(HOST_CURSOR QUIET x11 xfixes)
  if(HOST_CURSOR_FOUND)
    target_compile_definitions(vncsixel PRIVATE HAVE_HOST_XFIXES)
    target_include_directories(vncsixel PRIVATE ${HOST_CURSOR_INCLUDE_DIRS})
    target_link_libraries(vncsixel PRIVATE ${CMAKE_DL_LIBS})
    message(STATUS "Host cursor: optional runtime XFixes backend")
  endif()
endif()
install(TARGETS vncsixel DESTINATION ${CMAKE_INSTALL_FULL_BINDIR})

if(BUILD_TESTING AND GTest_FOUND)
  include(GoogleTest)
  enable_testing()
  add_executable(vncsixel-input-tests tests/HelpTest.cxx tests/KeyQueueTest.cxx tests/InputTest.cxx tests/FramebufferTest.cxx tests/PaletteTest.cxx tests/DamageTest.cxx tests/FrameClockTest.cxx Input.cxx)
  target_include_directories(vncsixel-input-tests PRIVATE ${CMAKE_SOURCE_DIR}/common ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(vncsixel-input-tests PRIVATE GTest::gtest_main)
  target_compile_options(vncsixel-input-tests PRIVATE -std=gnu++17)
  gtest_discover_tests(vncsixel-input-tests)
endif()
